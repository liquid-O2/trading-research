f_htf_calculations(int l5, int l50, int l200, int sLen2, int sSmooth) =>
    sqa27_raw = ta.percentile_linear_interpolation(close, sLen2, 50.0)
    sqa27_smoothed = ta.sma(sqa27_raw, sSmooth)
    s5   = ta.sma(close, l5)
    s50  = ta.sma(close, l50)
    s200 = ta.sma(close, l200)
    c1 = close > s200 ? 1 : close < s200 ? -1 : 0
    c3 = sqa27_smoothed > s50 ? 1 : sqa27_smoothed < s50 ? -1 : 0
    c6 = s5 > s200 ? 1 : s5 < s200 ? -1 : 0
    [s200, s50, sqa27_smoothed, c1, c3, c6]

// ─── Delta zones — k-means clustering, drawn as zone EDGES ──────────────────
f_fmt(float v) =>
    a = math.abs(v)
    a >= 1e9 ? str.tostring(v/1e9, "#.#")+"B" : a >= 1e6 ? str.tostring(v/1e6, "#.#")+"M" : a >= 1e3 ? str.tostring(v/1e3, "#.#")+"K" : str.tostring(v, "#")

atr14    = ta.atr(14)
minZoneH = nz(atr14) * zonePadAtr

// Weighted 1-D k-means
f_kmeans(array<float> pxArr, array<float> wArr, array<int> tArr, int k, int iters) =>
    n     = array.size(pxArr)
    cCtr  = array.new<float>(k, na)
    cMin  = array.new<float>(k, na)
    cMax  = array.new<float>(k, na)
    cWsum = array.new<float>(k, 0.0)
    cCnt  = array.new<int>(k, 0)
    cT    = array.new<int>(k, na)
    if n > 0
        keff = math.min(k, n)
        srt  = array.copy(pxArr)
        array.sort(srt, order.ascending)
        for j = 0 to keff - 1
            idx = math.min(int((j + 0.5) / keff * n), n - 1)
            array.set(cCtr, j, array.get(srt, idx))
        sumW  = array.new<float>(keff, 0.0)
        sumWX = array.new<float>(keff, 0.0)
        for it = 1 to iters
            array.fill(sumW, 0.0)
            array.fill(sumWX, 0.0)
            for p = 0 to n - 1
                px    = array.get(pxArr, p)
                best  = 0
                bestD = math.abs(px - array.get(cCtr, 0))
                for c = 1 to keff - 1
                    d = math.abs(px - array.get(cCtr, c))
                    if d < bestD
                        bestD := d
                        best  := c
                w = array.get(wArr, p)
                array.set(sumW,  best, array.get(sumW,  best) + w)
                array.set(sumWX, best, array.get(sumWX, best) + w * px)
            for c = 0 to keff - 1
                sw = array.get(sumW, c)
                if sw > 0
                    array.set(cCtr, c, array.get(sumWX, c) / sw)
        for p = 0 to n - 1
            px    = array.get(pxArr, p)
            best  = 0
            bestD = math.abs(px - array.get(cCtr, 0))
            for c = 1 to keff - 1
                d = math.abs(px - array.get(cCtr, c))
                if d < bestD
                    bestD := d
                    best  := c
            w  = array.get(wArr, p)
            ti = array.get(tArr, p)
            array.set(cWsum, best, array.get(cWsum, best) + w)
            array.set(cCnt,  best, array.get(cCnt,  best) + 1)
            pmn = array.get(cMin, best)
            pmx = array.get(cMax, best)
            array.set(cMin, best, na(pmn) ? px : math.min(pmn, px))
            array.set(cMax, best, na(pmx) ? px : math.max(pmx, px))
            pct = array.get(cT, best)
            array.set(cT, best, na(pct) ? ti : math.min(pct, ti))
    [cMin, cMax, cWsum, cCnt, cT]

f_renderEdges(array<line> lines, array<label> labs, array<float> bMin, array<float> bMax, array<float> bW, array<int> bCnt, array<int> bT, array<float> rMin, array<float> rMax, array<float> rW, array<int> rCnt, array<int> rT) =>
    nLine = array.size(lines)
    nLab  = array.size(labs)
    for i = 0 to nLine - 1
        line.set_color(array.get(lines, i), color.new(color.gray, 100))
    for i = 0 to nLab - 1
        label.set_color(array.get(labs, i), color.new(color.gray, 100))
        label.set_textcolor(array.get(labs, i), color.new(color.gray, 100))
        label.set_text(array.get(labs, i), "")
    if showDeltaAreas
        k = array.size(bMin)
        maxW = 0.0
        for c = 0 to k - 1
            if array.get(bCnt, c) > 0
                maxW := math.max(maxW, array.get(bW, c))
            if array.get(rCnt, c) > 0
                maxW := math.max(maxW, array.get(rW, c))
        li  = 0
        lbi = 0
        for c = 0 to k - 1
            if array.get(bCnt, c) > 0 and li <= nLine - 2 and (maxW > 0 ? array.get(bW, c) / maxW : 1.0) * 100 >= minZonePct
                lo = array.get(bMin, c)
                hi = array.get(bMax, c)
                md = (lo + hi) / 2.0
                hf = math.max((hi - lo) / 2.0, minZoneH / 2.0)
                w  = array.get(bW, c)
                ratio  = maxW > 0 ? w / maxW : 1.0
                transp = math.min(math.max(int(heatColdTransp - (heatColdTransp - heatHotTransp) * ratio), 0), 100)
                wd  = ratio >= 0.66 ? 3 : ratio >= 0.33 ? 2 : 1
                x1  = array.get(bT, c)
                col = color.new(color.lime, transp)
                lnT = array.get(lines, li)
                lnB = array.get(lines, li + 1)
                line.set_xy1(lnT, x1, md + hf)
                line.set_xy2(lnT, time, md + hf)
                line.set_color(lnT, col)
                line.set_width(lnT, wd)
                line.set_xy1(lnB, x1, md - hf)
                line.set_xy2(lnB, time, md - hf)
                line.set_color(lnB, col)
                line.set_width(lnB, wd)
                li += 2
                if showZoneLabels and lbi <= nLab - 1
                    lb = array.get(labs, lbi)
                    label.set_xy(lb, time, md + hf)
                    label.set_text(lb, f_fmt(w))
                    label.set_color(lb, color.new(color.black, 100))
                    label.set_textcolor(lb, color.new(color.lime, math.max(transp - 25, 0)))
                    lbi += 1
            if array.get(rCnt, c) > 0 and li <= nLine - 2 and (maxW > 0 ? array.get(rW, c) / maxW : 1.0) * 100 >= minZonePct
                lo = array.get(rMin, c)
                hi = array.get(rMax, c)
                md = (lo + hi) / 2.0
                hf = math.max((hi - lo) / 2.0, minZoneH / 2.0)
                w  = array.get(rW, c)
                ratio  = maxW > 0 ? w / maxW : 1.0
                transp = math.min(math.max(int(heatColdTransp - (heatColdTransp - heatHotTransp) * ratio), 0), 100)
                wd  = ratio >= 0.66 ? 3 : ratio >= 0.33 ? 2 : 1
                x1  = array.get(rT, c)
                col = color.new(color.red, transp)
                lnT = array.get(lines, li)
                lnB = array.get(lines, li + 1)
                line.set_xy1(lnT, x1, md + hf)
                line.set_xy2(lnT, time, md + hf)
                line.set_color(lnT, col)
                line.set_width(lnT, wd)
                line.set_xy1(lnB, x1, md - hf)
                line.set_xy2(lnB, time, md - hf)
                line.set_color(lnB, col)
                line.set_width(lnB, wd)
                li += 2
                if showZoneLabels and lbi <= nLab - 1
                    lb = array.get(labs, lbi)
                    label.set_xy(lb, time, md - hf)
                    label.set_text(lb, f_fmt(w))
                    label.set_color(lb, color.new(color.black, 100))
                    label.set_textcolor(lb, color.new(color.red, math.max(transp - 25, 0)))
                    lbi += 1

f_collectLevels(array<float> bMin, array<float> bMax, array<float> bW, array<int> bCnt, array<float> rMin, array<float> rMax, array<float> rW, array<int> rCnt, float minH) =>
    lv = array.new<float>()
    k  = array.size(bMin)
    maxW = 0.0
    for c = 0 to k - 1
        if array.get(bCnt, c) > 0
            maxW := math.max(maxW, array.get(bW, c))
        if array.get(rCnt, c) > 0
            maxW := math.max(maxW, array.get(rW, c))
    for c = 0 to k - 1
        if array.get(bCnt, c) > 0 and (maxW > 0 ? array.get(bW, c) / maxW : 1.0) * 100 >= minZonePct
            md = (array.get(bMin, c) + array.get(bMax, c)) / 2.0
            hf = math.max((array.get(bMax, c) - array.get(bMin, c)) / 2.0, minH / 2.0)
            array.push(lv, md + hf)
            array.push(lv, md - hf)
        if array.get(rCnt, c) > 0 and (maxW > 0 ? array.get(rW, c) / maxW : 1.0) * 100 >= minZonePct
            md = (array.get(rMin, c) + array.get(rMax, c)) / 2.0
            hf = math.max((array.get(rMax, c) - array.get(rMin, c)) / 2.0, minH / 2.0)
            array.push(lv, md + hf)
            array.push(lv, md - hf)
    lv

// Stores
var array<float> bullPx   = array.new<float>()
var array<float> bullW    = array.new<float>()
var array<int>   bullTime = array.new<int>()
var array<float> bearPx   = array.new<float>()
var array<float> bearW    = array.new<float>()
var array<int>   bearTime = array.new<int>()

// Line/label pool
var array<line>   heatLines  = array.new<line>()
var array<label> heatLabels = array.new<label>()
if barstate.isfirst
    for i = 0 to 4 * kClusters - 1
        array.push(heatLines, line.new(time, close, time, close, xloc=xloc.bar_time, color=color.new(color.gray, 100), width=1))
    for i = 0 to 2 * kClusters - 1
        array.push(heatLabels, label.new(time, close, "", xloc=xloc.bar_time, style=label.style_label_left, color=color.new(color.gray, 100), textcolor=color.new(color.gray, 100), size=size.small))

// Cache resultados
var array<float> bullCMin = array.new<float>(kClusters, na)
var array<float> bullCMax = array.new<float>(kClusters, na)
var array<float> bullCW   = array.new<float>(kClusters, 0.0)
var array<int>   bullCCnt = array.new<int>(kClusters, 0)
var array<int>   bullCT   = array.new<int>(kClusters, na)
var array<float> bearCMin = array.new<float>(kClusters, na)
var array<float> bearCMax = array.new<float>(kClusters, na)
var array<float> bearCW   = array.new<float>(kClusters, 0.0)
var array<int>   bearCCnt = array.new<int>(kClusters, 0)
var array<int>   bearCT   = array.new<int>(kClusters, na)

detBullEvt = inSession and deltaBull
detBearEvt = inSession and deltaBear

if detBullEvt
    array.push(bullPx,   math.min(open, close))
    array.push(bullW,    math.abs(barDelta))
    array.push(bullTime, time)
    if array.size(bullPx) > maxDet
        array.shift(bullPx)
        array.shift(bullW)
        array.shift(bullTime)
    if showDeltaAreas
        [cmn, cmx, cw, ccnt, ct] = f_kmeans(bullPx, bullW, bullTime, kClusters, kmIter)
        bullCMin := cmn
        bullCMax := cmx
        bullCW   := cw
        bullCCnt := ccnt
        bullCT   := ct

if detBearEvt
    array.push(bearPx,   math.max(open, close))
    array.push(bearW,    math.abs(barDelta))
    array.push(bearTime, time)
    if array.size(bearPx) > maxDet
        array.shift(bearPx)
        array.shift(bearW)
        array.shift(bearTime)
    if showDeltaAreas
        [cmn, cmx, cw, ccnt, ct] = f_kmeans(bearPx, bearW, bearTime, kClusters, kmIter)
        bearCMin := cmn
        bearCMax := cmx
        bearCW   := cw
        bearCCnt := ccnt
        bearCT   := ct

f_renderEdges(heatLines, heatLabels, bullCMin, bullCMax, bullCW, bullCCnt, bullCT, bearCMin, bearCMax, bearCW, bearCCnt, bearCT)

// ─── Condition 4: Delta-line break / rejection (CORREGIDO) ──────────────────
// ARREGLO 1: Antes, si en la misma vela había rotura alcista Y rechazo bajista
//   a la vez, c4dir se quedaba CONGELADO. Ahora lo último que pasa manda.
// ARREGLO 2: Antes el breakout comparaba close[2] contra un nivel que se
//   recalcula. Ahora el cruce usa solo 1 vela de referencia (close[1]).
// ARREGLO 3: Antes el rebote se disparaba en CADA vela. Ahora exige que la
//   mecha entre al nivel SOLO en esta vela.
c4Levels = f_collectLevels(bullCMin, bullCMax, bullCW, bullCCnt, bearCMin, bearCMax, bearCW, bearCCnt, minZoneH)
if array.size(c4Levels) > 0
    for i = 0 to array.size(c4Levels) - 1
        L = array.get(c4Levels, i)
        // BREAKOUT al alza  o  REJECTION/rebote al alza  →  alcista
        if (close > L and close[1] <= L) or (close > L and low <= L and low[1] > L)
            c4dir := 1
        // BREAKOUT a la baja  o  REJECTION al alza  →  bajista
        if (close < L and close[1] >= L) or (close < L and high >= L and high[1] < L)
            c4dir := -1

// ─── Condition 5: Multi-Timeframe Confirmation (MTF) ────────────────────────
// C7 ahora es direccional (igual que C4) y SÍ participa en el régimen local.
currentLocalBull = (c1Bull ? 1 : 0) + (c2Bull ? 1 : 0) + (c3Bull ? 1 : 0) + (c4dir == 1 ? 1 : 0) + (c6Bull ? 1 : 0) + (c7Bull ? 1 : 0)
currentLocalBear = (c1Bear ? 1 : 0) + (c2Bear ? 1 : 0) + (c3Bear ? 1 : 0) + (c4dir == -1 ? 1 : 0) + (c6Bear ? 1 : 0) + (c7Bear ? 1 : 0)
estLocalRegime   = currentLocalBull >= currentLocalBear ? 1 : -1

mtfRegime = request.security(syminfo.tickerid, mtf_conf, estLocalRegime[1], barmerge.gaps_off, barmerge.lookahead_off)

c5Bull = mtfRegime == 1
c5Bear = mtfRegime == -1

// ─── Conteo de Votos Totales (Máximo 7) ──────────────────────────────────────
countBull = (c1Bull ? 1 : 0) + (c2Bull ? 1 : 0) + (c3Bull ? 1 : 0) + (c4dir == 1 ? 1 : 0) + (c5Bull ? 1 : 0) + (c6Bull ? 1 : 0) + (c7Bull ? 1 : 0)
countBear = (c1Bear ? 1 : 0) + (c2Bear ? 1 : 0) + (c3Bear ? 1 : 0) + (c4dir == -1 ? 1 : 0) + (c5Bear ? 1 : 0) + (c6Bear ? 1 : 0) + (c7Bear ? 1 : 0)

// ─── Algoritmo de Histéresis (Filtro Anti-Efecto Sierra) ─────────────────────
var int regime = -1 // Inicialización por defecto en Bearish

if regime == -1 and countBull >= minVotesBull
    regime := 1  // Solo pasa a Bullish si supera el umbral exigente de compras
else if regime == 1 and countBear >= minVotesBear
    regime := -1 // Solo pasa a Bearish si los bajistas recuperan con fuerza el umbral exigente

// Asignación limpia de colores según el sesgo directo amortiguado
color biasColor = regime == 1 ? color.new(color.lime, 90) : color.new(color.red, 90)

// ─── Plots (visible during London+NY session, or 24h with mode24h) ─────────
plot(showSma and inSession ? sma200htf : na, "SMA 200 HTF", color=c1Bull ? color.lime : c1Bear ? color.red : color.gray, linewidth=2)
plot(showSma and inSession ? sma50htf : na, "SMA 50 HTF", color=color.orange, linewidth=1)
plot(showSma and inSession ? sqa27htf : na, "SQA 27 HTF", color=color.white, linewidth=2)
bgcolor(showBg and inSession ? biasColor : na)

// Votos expuestos en la Data Window
plot(countBull, "Total BULL VOTE Count", color=color.green, display=display.data_window)
plot(countBear, "Total BEAR VOTE Count", color=color.red, display=display.data_window)
plot(c1dir, "C1 · Close vs SMA200", color=color.green, display=display.data_window)
plot(c2Bull ? 1 : c2Bear ? -1 : 0, "C2 · Delta Flow Vol", color=color.aqua, display=display.data_window)
plot(c3dir, "C3 · SQA27 vs SMA50 vote", color=color.purple, display=display.data_window)
plot(c4dir, "C4 · delta-line vote", color=color.gray, display=display.data_window)
plot(c5Bull ? 1 : c5Bear ? -1 : 0, "C5 · MTF confirmation vote", color=color.orange, display=display.data_window)
plot(c6dir, "C6 · SMA5 vs SMA200 vote", color=color.blue, display=display.data_window)
plot(c7Bull ? 1 : c7Bear ? -1 : 0, "C7 · Volume Anomaly break/reject", color=color.yellow, display=display.data_window)

// ─── Delta bubbles ─────────────────────────────────────────────────────────
f_size(float v) =>
    r = effThresh > 0 ? math.abs(v) / effThresh : 1.0
    r >= 3 ? size.large : r >= 1.8 ? size.normal : size.small

if showBubbles and deltaBull and inSession
    label.new(bar_index, low,  f_fmt(barDelta), yloc=yloc.belowbar, style=label.style_circle, color=color.new(color.lime,15), textcolor=color.black, size=f_size(barDelta))
if showBubbles and deltaBear and inSession
    label.new(bar_index, high, f_fmt(barDelta), yloc=yloc.abovebar, style=label.style_circle, color=color.new(color.red,15), textcolor=color.white, size=f_size(barDelta))

alertcondition(regime == 1 and regime[1] != 1, "Switch to BULLISH", "Bias: Hysteresis BULLISH regime")
alertcondition(regime == -1 and regime[1] != -1, "Switch to BEARISH", "Bias: Hysteresis BEARISH regime")
